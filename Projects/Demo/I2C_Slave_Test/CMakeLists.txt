# =============================================================================
# I2C_Slave_Test - 硬件 I2C 从机模式测试
# =============================================================================
# 演示 S300 作为 I2C 从机设备:
# - 模拟寄存器式从设备 (类似 EEPROM/传感器)
# - 支持主机读写操作
# - 中断驱动的事件处理
#
# 测试方法:
# - 使用另一块 S300 或外部 I2C 主机连接
# - 或用 I2C_Hardware_Test Demo 作为主机测试
# =============================================================================

cmake_minimum_required(VERSION 3.16)
project(I2C_Slave_Test)

# 使用硬件 I2C 驱动
s300_add_executable(
    TARGET s300_i2c_slave_test
    SOURCES
    Src/main.c
    DRIVERS uart i2c
    DBG_NAME i2c_slv
)
